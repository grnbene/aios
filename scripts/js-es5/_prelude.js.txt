function assert(c, m) { if (!c) throw new Error("assertion failed" + (m ? ": " + m : "")); }
function eq(a, b, m) { if (!(a === b || (a !== a && b !== b))) throw new Error("expected " + String(b) + " got " + String(a) + (m ? " (" + m + ")" : "")); }
function throws(fn, C, m) { var ok = false; try { fn(); } catch (e) { ok = C ? e instanceof C : true; if (!ok) throw new Error("wrong error: " + e); } if (!ok) throw new Error("did not throw" + (m ? ": " + m : "")); }
