# A static library reached by several paths (aios, through aios_git_cli and aios_sh_lib) is named once on the link
# line where the linker needs it once; Apple's ld warned "ignoring duplicate libraries" for every executable.
if(POLICY CMP0156)
  cmake_policy(SET CMP0156 NEW)
endif()
# On macOS the libraries taken from Homebrew (zstd, libdeflate) are built for the macOS they were installed on, so a
# binary that links them runs on that macOS and later, whatever older default the SDK names; the linker said so once
# per object. Say the same up front. -DCMAKE_OSX_DEPLOYMENT_TARGET or MACOSX_DEPLOYMENT_TARGET still decides.
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET AND NOT DEFINED ENV{MACOSX_DEPLOYMENT_TARGET})
  execute_process(COMMAND sw_vers -productVersion OUTPUT_VARIABLE AIOS_MACOS_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
  string(REGEX MATCH "^[0-9]+" AIOS_MACOS_MAJOR "${AIOS_MACOS_VERSION}")
  if(AIOS_MACOS_MAJOR)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "${AIOS_MACOS_MAJOR}.0" CACHE STRING "Minimum macOS the binaries run on" FORCE)
  endif()
endif()

# AIOS — a software computer architecture. Phase 1 runs on a Linux host; the
# host is reached ONLY through hal/ so it can be removed later.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# A fully static link (-static, as build-static is configured): static glibc cannot load host libraries or use the
# host's NSS (dlopen, getaddrinfo, getpwuid_r warn at link time and misbehave on a host with another glibc). The
# HAL then loads no libraries and resolves names and accounts itself.
if(CMAKE_EXE_LINKER_FLAGS MATCHES "(^| )-static( |$)")
  add_compile_definitions(AIOS_STATIC_LINK=1)
endif()
if(MSVC)
  # sql_database.cpp alone is 28k+ lines of template-heavy SQL evaluation
  # code: its object file's section count overflows the COFF 32-bit limit
  # without this.
  add_compile_options(/W3 /bigobj)
  # Static, same as Linux's -static and macOS's "system libraries only": the
  # kernel bundle ships a binary that runs on a machine with nothing else on
  # it, not one that needs whatever VC++ redistributable happens to be
  # installed (the dynamic CRT's api-ms-win-crt-*.dll apiset forwarders did
  # not resolve for converge-engine.exe on this SDK/OS combination, though
  # oddly not for aios-agent-control.exe — not worth chasing further when
  # static sidesteps the whole class of problem).
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wno-unused-parameter)
  # Every executable links the whole of libaios.a, so without this the ASQL
  # engine and its two clients carried the git grammars, the compositor and the
  # font atlas (23 MB each, the console included). The linker drops what no
  # entry point or global constructor can reach; registries and function-pointer
  # tables are references, so they stay. -DAIOS_GC_SECTIONS=OFF turns it off.
  option(AIOS_GC_SECTIONS "Drop unreachable code and data at link time" ON)
  if(AIOS_GC_SECTIONS)
    if(APPLE)
      add_link_options(-Wl,-dead_strip)
    else()
      add_compile_options(-ffunction-sections -fdata-sections)
      add_link_options(-Wl,--gc-sections)
    endif()
  endif()
endif()

# The host boundary is picked here and nowhere else: exactly one platform's
# implementation of hal::Host / hal::net / hal::block is compiled in. Every
# line above hal/ is the same source on every host.
if(APPLE)
  list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/hal/(linux_|win_)")
  set(AIOS_HOST_BOUNDARY "macOS (src/hal/mac_*.cpp)")
  # Darwin keeps the accelerator runtime's ABI declarations in a framework and
  # spells them <OpenCL/cl.h>. The compute layer asks for the substrate-neutral
  # spelling, so this host's spelling is resolved here and not in that layer.
  file(WRITE ${CMAKE_BINARY_DIR}/host-compat/CL/cl.h
    "// Generated by CMake for the macOS host boundary.\n#define CL_SILENCE_DEPRECATION 1\n#include <OpenCL/cl.h>\n")
  set(AIOS_HOST_COMPAT_INCLUDE ${CMAKE_BINARY_DIR}/host-compat)
  # The microphone. src/hal/mac_audio.cpp drives CoreAudio's AudioQueue, which
  # is the substrate's real input path.
  find_library(AIOS_AUDIOTOOLBOX AudioToolbox REQUIRED)
  find_library(AIOS_COREAUDIO CoreAudio REQUIRED)
  find_library(AIOS_COREFOUNDATION CoreFoundation REQUIRED)
  set(AIOS_HOST_AUDIO_LIBS ${AIOS_AUDIOTOOLBOX} ${AIOS_COREAUDIO} ${AIOS_COREFOUNDATION})
elseif(UNIX)
  list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/hal/(mac_|win_)")
  set(AIOS_HOST_BOUNDARY "Linux (src/hal/linux_*.cpp)")
elseif(WIN32)
  list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/hal/(linux_|mac_)")
  # posix_process.cpp and posix_sync.cpp are the pthread/fork half of the
  # POSIX boundary; win_process.cpp and win_sync.cpp take their place.
  # posix_clipboard.cpp has no POSIX calls in it at all and needs no
  # replacement; posix_speech.cpp needs none either (its one unistd.h use is
  # guarded in the file itself).
  list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/hal/posix_(process|sync|tty|account)\\.cpp$")
  # Not ported, and confirmed (by actually linking aios-agent-control) that
  # nothing it reaches needs them: each is a self-contained optional feature,
  # not a hal:: boundary gap. (secp256k1 looked Bitcoin-only by that standard
  # too, until converge-engine's link — which reaches the chain/BTC/ETH
  # sidecar code through the SQL engine — showed otherwise: it has no
  # __int128 or POSIX calls of its own, so it just needed to be un-excluded.)
  #   compute/opencl_device.cpp                        needs the OpenCL SDK's CL/cl.h, not installed here
  #                                                     (opencl_device_stub.cpp gives register_opencl_devices
  #                                                     an empty, honest answer instead)
  #   userland/compositor.cpp                          termios raw-mode terminal input
  #   userland/mail_client.cpp                         raw BSD sockets, bypasses hal::net
  #   sidecar/{fs_topology,shm_abi}.cpp                 POSIX mmap/dirent shared-memory and fs-mirror ABI
  # curve25519.cpp, nistp.cpp and rsa.cpp (also __int128) and
  # intelligence/external_process.cpp (fork/pipe/waitpid) ARE needed — TLS's
  # x25519/ECDSA/RSA and agent_control_serve's --external-mind reach them
  # directly — so those are ported (win_int128.hpp; CreateProcess+CreatePipe)
  # rather than excluded. userland/{sql_database,sql_things,sql_cells}.cpp (the
  # ERP "Things" decimal type's exact arithmetic) are needed too — converge's
  # read_model.cpp reaches SqlDatabase/CellEngine directly — so __int128 there
  # is AIOS_INT128 (portable_int128.hpp: MSVC's own std::_Signed128, the same
  # type its <charconv> big-integer formatting uses, not a hand-rolled bignum).
  list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/compute/opencl_device\\.cpp$")
  list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/userland/compositor\\.cpp$")
  list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/sidecar/(fs_topology|shm_abi)\\.cpp$")
  # Not ported, same "confirmed by actually linking" standard as above:
  #   userland/sense.cpp                     POSIX posix_spawn (<spawn.h>)
  list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/userland/sense\\.cpp$")
  set(AIOS_HOST_BOUNDARY "Windows (src/hal/win_*.cpp)")
else()
  message(FATAL_ERROR "no hal:: implementation for this host; see src/hal/")
endif()
if(NOT WIN32)
  # The OpenCL SDK's CL/cl.h is optional on Linux and macOS too: a box without
  # it (a plain Ubuntu without opencl-headers) builds with the honest stub, as
  # Windows does, instead of failing on the one compute source.
  if(NOT APPLE)
    include(CheckIncludeFileCXX)
    check_include_file_cxx("CL/cl.h" AIOS_HAVE_CL_H)
  else()
    set(AIOS_HAVE_CL_H ON)
  endif()
  if(AIOS_HAVE_CL_H)
    list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/compute/opencl_device_stub\\.cpp$")
  else()
    message(STATUS "OpenCL headers not found (CL/cl.h): building with opencl_device_stub.cpp")
    list(FILTER AIOS_SOURCES EXCLUDE REGEX "/src/compute/opencl_device\\.cpp$")
  endif()
endif()

# tree-sitter (third_party/tree-sitter, MIT) and the grammars the structural merge reads
# (third_party/tree-sitter-grammars, each MIT): C sources compiled into libaios itself, so whatever links aios (the
# engine included) has them with no other library to find. Their own code: built quietly, as their authors ship it.
# Each grammar is compiled from parser_packed.c, not parser.c: the aios-ts-pack tool (tools/ts-pack) turns the parser's big
# tables (14 MB across the grammars) into zlib streams that src/userland/ts_packed.cpp unpacks on a grammar's first use.
# The generated files are checked in. When a parser.c changes, `cmake --build <build> --target ts-pack` remakes them
# (ctest's ts-pack.fresh fails until you do) and `--target ts-pack-verify` compares the packed grammars with the originals.
include(cmake/ts-grammars.cmake)
set(AIOS_TS_SOURCES third_party/tree-sitter/lib/src/lib.c)
foreach(grammar IN LISTS AIOS_TS_GRAMMARS)
  string(REPLACE "|" ";" row "${grammar}")
  list(GET row 1 ts_dir)
  list(GET row 3 ts_scanner)
  list(APPEND AIOS_TS_SOURCES third_party/tree-sitter-grammars/${ts_dir}/parser_packed.c)
  if(ts_scanner)
    list(APPEND AIOS_TS_SOURCES third_party/tree-sitter-grammars/${ts_dir}/scanner.c)
  endif()
endforeach()
if(MSVC)
  set_source_files_properties(${AIOS_TS_SOURCES} PROPERTIES COMPILE_OPTIONS "/w")
else()
  set_source_files_properties(${AIOS_TS_SOURCES} PROPERTIES COMPILE_OPTIONS "-w;-std=gnu11")
endif()
set_source_files_properties(third_party/tree-sitter-grammars/xml/xml/src/scanner.c PROPERTIES INCLUDE_DIRECTORIES
  "${CMAKE_CURRENT_SOURCE_DIR}/third_party/tree-sitter-grammars/xml/xml/src")   # its shared scanner.h includes tree_sitter/parser.h
set_source_files_properties(third_party/tree-sitter/lib/src/lib.c PROPERTIES INCLUDE_DIRECTORIES
  "${CMAKE_CURRENT_SOURCE_DIR}/third_party/tree-sitter/lib/include;${CMAKE_CURRENT_SOURCE_DIR}/third_party/tree-sitter/lib/src")
list(APPEND AIOS_SOURCES ${AIOS_TS_SOURCES})
add_library(aios STATIC ${AIOS_SOURCES})
target_include_directories(aios PRIVATE third_party/tree-sitter/lib/include)

# aios-ts-pack: not part of the normal build (it compiles every grammar's original parser.c a second time, under another
# name, to read its tables). Needs the aios library only for deflate and the language runtime.
set(AIOS_TS_PACK_GEN ${CMAKE_BINARY_DIR}/ts-pack)
set(AIOS_TS_PACK_ORIG "")
set(AIOS_TS_PACK_REGISTRY "")
foreach(grammar IN LISTS AIOS_TS_GRAMMARS)
  string(REPLACE "|" ";" row "${grammar}")
  list(GET row 0 ID)
  list(GET row 1 ts_dir)
  list(GET row 2 TSFN)
  set(PARSER ${CMAKE_SOURCE_DIR}/third_party/tree-sitter-grammars/${ts_dir}/parser.c)
  configure_file(tools/ts-pack/orig.c.in ${AIOS_TS_PACK_GEN}/orig_${ID}.c @ONLY)
  list(APPEND AIOS_TS_PACK_ORIG ${AIOS_TS_PACK_GEN}/orig_${ID}.c)
  string(APPEND AIOS_TS_PACK_REGISTRY "AIOS_TS_ENTRY(${ID}, \"${ts_dir}\", ${TSFN})\n")
endforeach()
file(WRITE ${AIOS_TS_PACK_GEN}/ts-registry.inc.new "${AIOS_TS_PACK_REGISTRY}")
configure_file(${AIOS_TS_PACK_GEN}/ts-registry.inc.new ${AIOS_TS_PACK_GEN}/ts-registry.inc COPYONLY)
add_executable(aios-ts-pack EXCLUDE_FROM_ALL tools/ts-pack/main.cpp tools/ts-pack/fields.c ${AIOS_TS_PACK_ORIG})
if(MSVC)
  set_source_files_properties(tools/ts-pack/fields.c ${AIOS_TS_PACK_ORIG} PROPERTIES COMPILE_OPTIONS "/w")
else()
  set_source_files_properties(tools/ts-pack/fields.c ${AIOS_TS_PACK_ORIG} PROPERTIES COMPILE_OPTIONS "-w;-std=gnu11")
endif()
set_source_files_properties(tools/ts-pack/fields.c PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/tree-sitter/lib/src;${CMAKE_CURRENT_SOURCE_DIR}/third_party/tree-sitter/lib/include")
target_include_directories(aios-ts-pack PRIVATE tools/ts-pack ${AIOS_TS_PACK_GEN} third_party/tree-sitter/lib/include)
target_compile_definitions(aios-ts-pack PRIVATE AIOS_TS_SOURCE_ROOT="${CMAKE_SOURCE_DIR}")
target_link_libraries(aios-ts-pack PRIVATE aios)
add_custom_target(ts-pack COMMAND aios-ts-pack ${CMAKE_SOURCE_DIR} COMMENT "writing parser_packed.c for every grammar" VERBATIM)
add_custom_target(ts-pack-verify COMMAND aios-ts-pack --verify COMMENT "comparing the packed grammars with the originals" VERBATIM)
if(WIN32)
  target_compile_definitions(aios PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()
# AVX2/FMA distance kernels, only where the build host is an x86-64 that has
# them: an unconditional -mavx2 does not compile for arm64 (metal, the Pi) and
# faults with SIGILL on an x86 without AVX2. The code falls back to scalar.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64" AND EXISTS "/proc/cpuinfo")
  file(READ "/proc/cpuinfo" AIOS_CPUINFO)
  if(AIOS_CPUINFO MATCHES " avx2 " AND AIOS_CPUINFO MATCHES " fma ")
    set_source_files_properties(src/memory/hnsw_index.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
  endif()
endif()

  find_program(SDL2_CONFIG sdl2-config)
  if(SDL2_CONFIG)
    execute_process(COMMAND ${SDL2_CONFIG} --cflags OUTPUT_VARIABLE SDL2_CFLAGS OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND ${SDL2_CONFIG} --libs OUTPUT_VARIABLE SDL2_LDFLAGS OUTPUT_STRIP_TRAILING_WHITESPACE)
    separate_arguments(SDL2_CFLAGS_LIST NATIVE_COMMAND "${SDL2_CFLAGS}")
    separate_arguments(SDL2_LDFLAGS_LIST NATIVE_COMMAND "${SDL2_LDFLAGS}")
    set(SDL2_FOUND TRUE)
  endif()
endif()

  message(STATUS "AIOS Hardware Compositor: SDL2 enabled")

    target_include_directories(aios PUBLIC ${SDL2_INCLUDE_DIRS})

  elseif(SDL2_LDFLAGS_LIST)
    target_compile_options(aios PUBLIC ${SDL2_CFLAGS_LIST})
    target_link_libraries(aios PUBLIC ${SDL2_LDFLAGS_LIST})
  endif()

# aios-sh: the POSIX shell and utilities of aios (tools/aios-sh/sh.hpp), the shell of every agent command
# and check on Windows, Linux and macOS (on Windows because Git for Windows' bash cannot start at low
# integrity; elsewhere so all three run the same shell).
# Proof: node tools/proof/aios-sh/differential.mjs <aios-sh> runs every case through bash and aios-sh (the
# oracle is bash 5 with GNU utilities: macOS' /bin/bash 3.2 and BSD tools disagree with both).

  file(GLOB AIOS_SH_SOURCES CONFIGURE_DEPENDS tools/aios-sh/*.cpp)

  target_compile_definitions(aios-cli PRIVATE AIOS_HAVE_SH=1)
  target_link_libraries(aios-sh PRIVATE aios aios_git_cli)
  if(MSVC)
    # Every pipeline stage is a thread, and std::regex recurses on long lines: a deep stack for each.
    target_link_options(aios-sh PRIVATE /STACK:67108864)
    target_link_libraries(aios-sh PRIVATE shell32)
  endif()
endif()

    endif()
  endif()
endforeach()
